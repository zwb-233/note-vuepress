  • 将配置添加到~/.vimrc中，其中的YouCompleteMe插件需要使用Python编译和执行，请确认安装了python-dev-is-python3

    "通用配置

set tabstop=4  " Real tab width

set softtabstop=2  " How many spaces will expand from tab

set smartindent  " Ease editing code by smart auto indenting

set shiftwidth=2   " How many spaces for auto indent

syntax on  " 开启语法高亮

colorscheme peachpuff "vim配置方案

set number  " 显示行号

set hls "搜索时高亮显示被找到的文本

set scrolloff=3 " 上下可视行数

set incsearch   " 搜索时高亮显示被找到的文本

set ignorecase smartcase "搜索时默认不区分大小写，只有搜索关键字中出现一个大字母时才区分大小写

set enc=utf-8  "编码设置

" 自动去除尾行空格和tab

" From: Vigil <vim5632@rainslide.net>

function RemoveTrailingWhitespace()

    if &ft != "diff"

        let b:curcol = col(".")

        let b:curline = line(".")

        " 去掉每一尾行的空格和Tab

        silent! %s/\s\+$//

        " 去掉文件最后无内容的行尾

        " silent! %s/\(\s*\n\)\+\%$//

        call cursor(b:curline, b:curcol)

    endif

endfunction

  • 使用vim-plug管理和安装插件，插件管理器支持在vim界面输入下列命令

    • :PlugInstall：安装未下载的插件
    • :PlugUpdate：更新所有插件
    • :PlugClean：在删除了对应的插件引入配置Plug行后执行，移除插件
    • :PlugStatus：查找所有在配置中的插件状态
